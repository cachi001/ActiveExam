# p. ej. el import de Moodle XML (C-69). Sin esto FastAPI tira RuntimeError al arrancar.

# Cliente HTTP async para el write-back de nota a Moodle (REST WS core_grades_update_grades, C-69).

asyncpg==0.30.0

# falla con ModuleNotFoundError: psycopg2. La app usa asyncpg; Alembic usa psycopg2.

pyjwt[crypto]==2.10.1
# Hashing de passwords (C-55): bcrypt 12 rounds para el provider JWT propio.

passlib[bcrypt]==1.7.4
bcrypt<5.0
# Pillow: decodifica el screenshot (base64 -> RGB) para la re-inferencia MediaPipe.
# No es dependencia de mediapipe; el adapter la prefiere antes que opencv.

# Re-inferencia server-side con MediaPipe (mismo motor que el cliente, D8).
# Tradeoff: imagen Docker mas grande / cold start mas lento que opencv-python-headless,
# pero necesario para comparacion apples-to-apples cliente-vs-servidor (D8).
mediapipe>=0.10.14

# Export PDF de estadisticas institucionales (C-20). Python puro (sin libs de

# Render de graficos (PNG) que se embeben en el PDF. Ya viene transitivo de
# mediapipe, pero lo declaramos porque el export PDF lo usa DIRECTO (backend Agg).
